<script>
(async()=>{
 const base="/distendal/";
 try{
  const response=await fetch(base+"index.html",{cache:"no-store"});
  if(!response.ok)throw new Error("index.html tidak ditemukan");
  let html=await response.text();
  html=html.replace("<head>","<head><base href=\""+base+"\">");
  document.open();document.write(html);document.close();
 }catch(error){
  document.body.innerHTML="<h2>Halaman gagal dimuat</h2><p>Pastikan file index.html sudah ada di repository DISTENDAL.</p>";
 }
})();
</script></body></html>
